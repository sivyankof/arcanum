```python
# -*- coding: utf-8 -*-
"""Выжимка исходников для авторов: m5 — 16 придворных карт; m6 — контекст курса + карты примеров.
Запуск из корня репо: python extract.py m5 | m6 [ids для примеров М6 через запятую]"""
import io, json, sys
cards = {c["id"]: c for c in json.load(io.open("content/cards.json", encoding="utf-8"))["cards"]}
course = json.load(io.open("content/course.json", encoding="utf-8"))["modules"]
lessons = {l["id"]: l for m in course for l in m["lessons"]}
out = io.TextIOWrapper(sys.stdout.buffer, encoding="utf-8")

def card_block(cid, blocks=("general", "reversed", "symbolism")):
    c = cards[cid]
    print(f"### {cid} — {c['name']['ru']} / {c['name']['en']}", file=out)
    print(f"keywords ru: {', '.join(c['keywords']['ru'])}", file=out)
    print(f"keywords en: {', '.join(c['keywords']['en'])}", file=out)
    print(f"Уэйт (рисунок): {c['source']['waite_description']}", file=out)
    for b in blocks:
        print(f"[{b}.ru] {c['content'][b]['ru']}", file=out)
        print(f"[{b}.en] {c['content'][b]['en']}", file=out)
    print(file=out)

mode = sys.argv[1]
if mode == "m5":
    print("# Исходники М5 — 16 придворных карт (канон cards.json)\n", file=out)
    for lid in ("m5l1", "m5l2", "m5l3", "m5l4"):
        l = lessons[lid]
        print(f"## {lid} {l['title']['ru']} / {l['title']['en']}\n", file=out)
        for cid in l["cards"]:
            card_block(cid)
elif mode == "m6":
    print("# Контекст для авторов М6\n", file=out)
    for lid in ("m1l4", "m2l6", "m5l1", "m5l2", "m5l3", "m5l4"):
        l = lessons[lid]
        print(f"## {lid} {l['title']['ru']} / {l['title']['en']}\n", file=out)
        print(l["theory"]["ru"], "\n", file=out)
        print(l["theory"]["en"], "\n", file=out)
    ids = sys.argv[2].split(",") if len(sys.argv) > 2 else []
    if ids:
        print("## Карты для примеров\n", file=out)
        for cid in ids:
            card_block(cid, ("general", "reversed"))
```

- [ ] **Шаг 3: `$SP/assemble.py`** — валидация и раскладка. Точный код:

```python
# -*- coding: utf-8 -*-
"""Собирает уроки модуля из JSON-черновиков авторов в content/course.json (theory)
и content/quiz-<mod>.json (вопросы). Валидирует по спеке 26 (М5–М6). Идемпотентен.
Формат входа (файл на урок): {"lessonId": "m5l1", "theory": {"ru": "...", "en": "..."},
"questions": [5 × {"type": "single"|"card", "q": {ru,en}, "options": [{ru,en}×3],
"correct": 0..2, "explain": {ru,en}, "cardId"?: "w11"}]}
Запуск из корня репо: python assemble.py m5 m5/m5l1.json m5/m5l2.json ..."""
import io, json, re, sys
from collections import Counter

MOD = sys.argv[1]
FILES = sys.argv[2:]
LIMITS = {"m5": (1700, 1900, 2050), "m6": (1700, 2200, 2400)}  # ru_min, ru_max, en_max
PARA_MAX = 550
ROOT_COURSE = "content/course.json"
ROOT_QUIZ = f"content/quiz-{MOD}.json"

cards = {c["id"] for c in json.load(io.open("content/cards.json", encoding="utf-8"))["cards"]}
course = json.load(io.open(ROOT_COURSE, encoding="utf-8"))
lessons = {l["id"]: l for m in course["modules"] for l in m["lessons"]}
errors = []

def err(msg):
    errors.append(msg)

def check_theory(lid, th):
    ru_min, ru_max, en_max = LIMITS[MOD]
    ru, en = th.get("ru", ""), th.get("en", "")
    if not (ru_min <= len(ru) <= ru_max):
        err(f"{lid}: theory.ru {len(ru)} знаков, коридор {ru_min}–{ru_max}")
    if not (ru_min <= len(en) <= en_max):
        err(f"{lid}: theory.en {len(en)} знаков, коридор {ru_min}–{en_max}")
    pru, pen = ru.split("\n\n"), en.split("\n\n")
    if len(pru) != len(pen):
        err(f"{lid}: абзацев ru {len(pru)} ≠ en {len(pen)}")
    for lang, paras in (("ru", pru), ("en", pen)):
        for i, p in enumerate(paras, 1):
            if len(p) > PARA_MAX:
                err(f"{lid}: theory.{lang} абзац {i} — {len(p)} знаков (лимит {PARA_MAX})")
    if not pru[-1].startswith("Попробуйте сегодня"):
        err(f"{lid}: последний абзац ru не начинается «Попробуйте сегодня»")
    if not pen[-1].startswith("Try this today"):
        err(f"{lid}: последний абзац en не начинается «Try this today»")
    for lang, text in (("ru", ru), ("en", en)):
        if re.search(r"^\s|\s$|  ", text):
            err(f"{lid}: theory.{lang} — лишние пробелы в начале/конце или двойной пробел")
    if re.search(r"\bStrength\b|\bJudgement\b", en):
        err(f"{lid}: theory.en — неканоничное имя карты (канон: Fortitude, The Last Judgment)")
    if re.search(r"\b(color|colored|labor|honor|defense|center)\b", en):
        err(f"{lid}: theory.en — американское написание, канон британский")

def check_questions(lid, qs):
    if len(qs) != 5:
        err(f"{lid}: вопросов {len(qs)}, нужно 5"); return
    longest = 0
    for i, q in enumerate(qs, 1):
        w = f"{lid} в{i}"
        if q.get("type") not in ("single", "card"): err(f"{w}: type {q.get('type')!r}")
        if q.get("type") == "card" and q.get("cardId") not in cards: err(f"{w}: cardId {q.get('cardId')!r} нет в колоде")
        opts = q.get("options", [])
        if len(opts) != 3: err(f"{w}: вариантов {len(opts)}")
        if not (0 <= q.get("correct", -1) < 3): err(f"{w}: correct {q.get('correct')!r}")
        for f in ("q", "explain"):
            if not (q.get(f, {}).get("ru") and q.get(f, {}).get("en")): err(f"{w}: {f} не двуязычен")
        for o in opts:
            if not (o.get("ru") and o.get("en")): err(f"{w}: вариант не двуязычен")
        for lang in ("ru", "en"):
            texts = [o.get(lang, "").strip().lower() for o in opts]
            if len(set(texts)) != 3: err(f"{w}: варианты {lang} повторяются")
        if re.match(r"^(Верно|Да|Точно|Именно|Правильно)[!.,]", q.get("explain", {}).get("ru", "")):
            err(f"{w}: explain.ru начинается утвердительным зачином")
        if re.match(r"^(Correct|Yes|Exactly|Right|Indeed)[!.,]", q.get("explain", {}).get("en", "")):
            err(f"{w}: explain.en начинается утвердительным зачином")
        if q.get("type") == "single" and len(opts) == 3 and 0 <= q.get("correct", -1) < 3:
            lens = [len(o.get("ru", "")) for o in opts]
            if lens[q["correct"]] == max(lens) and lens.count(max(lens)) == 1:
                longest += 1
    if len({q.get("correct") for q in qs}) < 2:
        err(f"{lid}: все correct на одном индексе")
    if longest > 2:
        err(f"{lid}: верный вариант самый длинный в {longest} вопросах из 5 (порог 2)")

drafts = []
for path in FILES:
    d = json.load(io.open(path, encoding="utf-8"))
    lid = d.get("lessonId")
    if lid not in lessons or not lid.startswith(MOD):
        err(f"{path}: lessonId {lid!r} не из модуля {MOD}"); continue
    check_theory(lid, d.get("theory", {}))
    check_questions(lid, d.get("questions", []))
    drafts.append(d)

if errors:
    print("ОШИБКИ:\n  " + "\n  ".join(errors))
    sys.exit(1)

for d in drafts:
    lessons[d["lessonId"]]["theory"] = {"ru": d["theory"]["ru"], "en": d["theory"]["en"], "status": "draft"}
io.open(ROOT_COURSE, "w", encoding="utf-8", newline="\n").write(
    json.dumps(course, ensure_ascii=False, indent=1) + "\n")

quiz = {
    "_comment": f"Вопросы викторин {MOD.upper()}, черновик (статус draft). Схема: type single|card; card имеет cardId для показа изображения. Слияние в course.json — scripts/merge_quiz.py (берёт все content/quiz-*.json), статусы двигает scripts/set_status.py.",
    "status": "draft",
    "lessons": [{"lessonId": d["lessonId"], "questions": d["questions"]} for d in drafts],
}
io.open(ROOT_QUIZ, "w", encoding="utf-8", newline="\n").write(json.dumps(quiz, ensure_ascii=False, indent=1) + "\n")
print(f"OK: теория {len(drafts)} уроков → {ROOT_COURSE}; вопросы → {ROOT_QUIZ}. Дальше: python scripts/merge_quiz.py")
```

      Формат записи проверен при составлении плана: `json.dumps(course, ensure_ascii=False, indent=1) + "\n"`
      с LF воспроизводит текущий `content/course.json` байт в байт — после сборки `git diff --stat`
      обязан показывать ТОЛЬКО строки теории затронутых уроков. Если дифф «пересобрал весь файл» —
      что-то изменилось в файле, разобраться до коммита.

- [ ] **Шаг 4: `$SP/openers.py`** — зачины по языкам. Точный код:

```python
# -*- coding: utf-8 -*-
"""Повторы зачинов (первые два слова) — теорий и упражнений по всем модулям, вопросов внутри модуля;
отдельно ru и en. Порог: 3+. Запуск из корня репо."""
import io, json, re
from collections import Counter, defaultdict
course = json.load(io.open("content/course.json", encoding="utf-8"))["modules"]
PREFIX = {"ru": r"^Попробуйте сегодня[:\s—-]*", "en": r"^Try this today[:\s—-]*"}

def two(text):
    w = re.findall(r"[\w'’-]+", text)
    return " ".join(w[:2]).lower()

theory, exercise, questions = defaultdict(Counter), defaultdict(Counter), defaultdict(Counter)
where = defaultdict(list)
for m in course:
    for l in m["lessons"]:
        th = l.get("theory")
        if th:
            for lang in ("ru", "en"):
                paras = th[lang].split("\n\n")
                t = two(paras[0]); theory[lang][t] += 1; where[("theory", lang, t)].append(l["id"])
                ex = two(re.sub(PREFIX[lang], "", paras[-1])); exercise[lang][ex] += 1; where[("exercise", lang, ex)].append(l["id"])
        for q in l.get("quiz") or []:
            for lang in ("ru", "en"):
                t = two(q["q"][lang]); questions[(m["id"], lang)][t] += 1; where[("q", m["id"], lang, t)].append(l["id"])

def report(title, counter, key):
    hits = [(k, v) for k, v in counter.most_common() if v >= 3]
    print(f"=== {title}: {'—' if not hits else len(hits)}")
    for k, v in hits:
        print(f"    «{k}…» × {v} — {', '.join(where[key + (k,)])}")

for lang in ("ru", "en"):
    report(f"теории.{lang}", theory[lang], ("theory", lang))
    report(f"упражнения.{lang}", exercise[lang], ("exercise", lang))
for (mid, lang), c in sorted(questions.items()):
    report(f"вопросы {mid}.{lang}", c, ("q", mid, lang))
```

```
Ты пишешь ОДИН урок курса по таро для новичков в приложении Arcanum. Урок — про придворные карты
одного ранга во всех четырёх мастях (4 карты). Отдай РОВНО один JSON-файл по схеме ниже, без пояснений.

ОСЬ МОДУЛЯ (обязательна, все четыре урока М5 стоят на ней):
ранг = зрелость роли: Паж — учится и приносит весть; Рыцарь — действует и движется; Королева —
владеет стихией внутри (inner mastery); Король — распоряжается ею вовне (outer mastery).
масть = стихия (курс уже прошёл это в М4): Жезлы — огонь/действие, Кубки — вода/чувства,

Двойное чтение: каждая придворная — «человек ИЛИ ваша роль/сторона» (в выжимке: «Как человек…
Как энергия/совет…»). Дай одно короткое правило, как понять, о ком речь: если вопрос про
отношения с кем-то — скорее человек; если про себя и решение — скорее ваша роль; сомневаетесь —
читайте как роль: это всегда полезно и никого не обвиняет.
НЕ учить: «элемент в элементе» Золотой Зари; соответствия «масть = цвет волос/глаз»; выбор

персонажа») и с М4 (стихии);
```
